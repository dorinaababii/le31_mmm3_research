```sql
CREATE TABLE self_hosted_ai_worker_coordination (
    id SERIAL PRIMARY KEY,
    name TEXT NOT NULL,
    config JSON NOT NULL,         -- FastAPI + React + SQLite + MCP + multi-agent-systems + claude + codex config
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

);

    id SERIAL PRIMARY KEY,
    coord_id INT NOT NULL REFERENCES self_hosted_ai_worker_coordination(id),
    name TEXT NOT NULL,
    type TEXT NOT NULL,           -- 'waiter-ui-agent' | 'cook-bot-agent' | 'owner-recap-agent' | 'claude' | 'codex' | ...
    device TEXT,                  -- which device the AI worker is running on
    model TEXT,                   -- which model the AI worker is using
    status TEXT NOT NULL DEFAULT 'active',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE per_task_graph (
    id SERIAL PRIMARY KEY,
    worker_id INT NOT NULL REFERENCES ai_worker(id),
    task_name TEXT NOT NULL,
    steps JSON NOT NULL,          -- ordered list of {step, action, params, status}
    status TEXT NOT NULL DEFAULT 'pending',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- append-only event log (the §3.3 append-only-ledger invariant)
CREATE TABLE progress_receipt (
    id BIGSERIAL PRIMARY KEY,
    task_id INT NOT NULL REFERENCES per_task_graph(id),
    step_index INT NOT NULL,
    status TEXT NOT NULL,         -- 'started' | 'completed' | 'failed'
    payload JSON NOT NULL,
    source TEXT,                  -- which file/URL/MCP-tool the step consumed (source-aware-usage)
    device TEXT,                  -- which device the step ran on (device-timeline)
    model TEXT,                   -- which model the step used (model-timeline)
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
```
