```sql
-- hypothetical future v2 surface (NOT IMPLEMENTED TODAY)
-- Multi-shop + per-shop + multi-shop-rollup

CREATE TABLE shop (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    address TEXT,
    vat_id TEXT,
    active INTEGER NOT NULL DEFAULT 1,
    created_at TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE customer (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    shop_id INTEGER NOT NULL REFERENCES shop(id),  -- per-shop-customer
    phone TEXT,
    name TEXT,
    dietary_restrictions TEXT,
    created_at TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE product (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    sku TEXT NOT NULL,
    name TEXT NOT NULL,
    category TEXT,
    price_eur NUMERIC(10, 2) NOT NULL,
    active INTEGER NOT NULL DEFAULT 1,
    UNIQUE (sku, shop_id),  -- per-shop SKU
    FOREIGN KEY (shop_id) REFERENCES shop(id)
);

CREATE TABLE sale (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    shop_id INTEGER NOT NULL REFERENCES shop(id),  -- per-shop-sale
    customer_id INTEGER REFERENCES customer(id),
    total_eur NUMERIC(10, 2) NOT NULL,
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE sale_item (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    sale_id INTEGER NOT NULL REFERENCES sale(id),
    product_id INTEGER NOT NULL REFERENCES product(id),
    quantity INTEGER NOT NULL,
    unit_price_eur NUMERIC(10, 2) NOT NULL
);

-- Multi-shop-rollup query
SELECT
    s.id AS shop_id,
    s.name AS shop_name,
    COUNT(DISTINCT sa.id) AS total_sales,
    SUM(sa.total_eur) AS total_revenue_eur
FROM shop s
LEFT JOIN sale sa ON sa.shop_id = s.id
GROUP BY s.id, s.name;
```
